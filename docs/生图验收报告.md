## 6. 驱动脚本全文（备份）

```js
import { spawnSync } from "node:child_process";
import { createRequire } from "node:module";
import { readFileSync, existsSync, mkdirSync } from "node:fs";
import path from "node:path";

const require2 = createRequire("D:/HEXO/package.json");
const sharp = require2("sharp");

const SKILL = "C:/Users/LENOVO/.agents/skills/tokenfree-image2";
const WORK = "C:/Users/LENOVO/AppData/Local/Temp/hexo-img-work";
const STAGE = path.join(WORK, "stage");
const OUT = "D:/HEXO/public/bg";
mkdirSync(STAGE, { recursive: true });
mkdirSync(OUT, { recursive: true });

const jobs = [
	{ name: "bg-eva", promptFile: "eva-bg.txt", kind: "bg" },
	{ name: "banner-eva", promptFile: "eva-banner.txt", kind: "banner" },
	{ name: "bg-retro90", promptFile: "retro90-bg.txt", kind: "bg" },
	{ name: "banner-retro90", promptFile: "retro90-banner.txt", kind: "banner" },
	{ name: "bg-sakura", promptFile: "sakura-bg.txt", kind: "bg" },
	{ name: "banner-sakura", promptFile: "sakura-banner.txt", kind: "banner" },
];

for (const job of jobs) {
	const dest = path.join(OUT, job.name + ".jpg");
	if (existsSync(dest)) {
		console.log("SKIP", job.name, "(already exists)");
		continue;
	}
	const prompt = readFileSync(path.join(WORK, job.promptFile), "utf8").trim();
	console.log("=== generating", job.name);
	const r = spawnSync(
		"python",
		[
			SKILL + "/scripts/generate_image.py",
			"--prompt", prompt,
			"--size", "3840x2160",
			"--output-dir", STAGE,
		],
		{ encoding: "utf8", maxBuffer: 16 * 1024 * 1024 },
	);
	const out = (r.stdout || "") + (r.stderr || "");
	let jsonLine = null;
	for (const line of out.split("\n")) {
		const t = line.trim();
		if (t.startsWith("{") && t.includes("output_path")) {
			try { jsonLine = JSON.parse(t); } catch { /* ignore */ }
		}
	}
	if (!jsonLine || !jsonLine.output_path || !existsSync(jsonLine.output_path)) {
		console.log("!!! FAILED", job.name, "rc=" + r.status, out.slice(-500));
		continue;
	}
	const src = jsonLine.output_path;
	try {
		if (job.kind === "bg") {
			await sharp(src)
				.resize(2560, 1440, { fit: "cover", position: "centre" })
				.jpeg({ quality: 82, progressive: true, mozjpeg: true })
				.toFile(dest);
		} else {
			// center-crop a 21:9 band from the 3840x2160 source, then resize
			const bandH = Math.round((3840 * 9) / 21); // 1646
			const top = Math.round((2160 - bandH) / 2);
			await sharp(src)
				.extract({ left: 0, top, width: 3840, height: bandH })
				.resize(2560, 1097)
				.jpeg({ quality: 82, progressive: true, mozjpeg: true })
				.toFile(dest);
		}
		console.log("OK", job.name, "->", dest);
	} catch (e) {
		console.log("!!! CONVERT FAILED", job.name, e.message);
	}
}
console.log("ALL DONE");
```
